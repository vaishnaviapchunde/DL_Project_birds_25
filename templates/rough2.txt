        document.getElementById('uploadForm').onsubmit = function(event) {
            event.preventDefault();
            const fileInput = document.getElementById('file');
            const file = fileInput.files[0];
            const reader = new FileReader();

            reader.onload = function(e) {
                const imageUrl = e.target.result;
                // Mock fetch call
                setTimeout(() => {
                    // Example prediction
                    const prediction = "Hoopoe";
                    document.getElementById("uploadedImage").src = imageUrl;
                    document.getElementById("uploadedImage").style.display = "block";
                    document.getElementById("prediction").textContent = `Prediction: ${prediction}`;
                    document.getElementById("scientificName").textContent = "Scientific Name: Upupa epops";
                    document.getElementById("description").textContent = "Hoopoes are colorful birds found across Afro-Eurasia. Known for their distinctive crown of feathers.";
                }, 1000);
            };

            reader.readAsDataURL(file);
        };
